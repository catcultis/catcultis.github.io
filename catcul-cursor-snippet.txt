<script id="catcul-cursor-script">
(() => {
  'use strict';
  // Independent of Supabase and authentication. Safe to include once on each page.
  if (document.getElementById('catcul-cursor')) return;
  const root = document.documentElement;
  const pointerMedia = matchMedia('(any-hover: hover) and (any-pointer: fine)');
  const contrastMedia = matchMedia('(forced-colors: active)');
  const storageKey = 'catcul.customCursor';
  let enabled = true;
  try { enabled = localStorage.getItem(storageKey) !== 'off'; } catch (_) {}
  let visible = false, dragging = false, x = 0, y = 0, frame = 0;
  const cursor = document.createElement('div');
  cursor.id = 'catcul-cursor';
  cursor.setAttribute('aria-hidden', 'true');
  cursor.innerHTML = '<span class="cc-halo"></span><span class="cc-dot"></span><span class="cc-symbol cc-arrow"><svg viewBox="0 0 24 24"><path d="M5 12h14m-6-6 6 6-6 6"/></svg></span><span class="cc-symbol cc-plus"><svg viewBox="0 0 24 24"><path d="M12 5v14M5 12h14"/></svg></span><span class="cc-symbol cc-stop"><svg viewBox="0 0 24 24"><circle cx="12" cy="12" r="8"/><path d="m6.5 6.5 11 11"/></svg></span><span class="cc-caret"></span>';
  const toggle = document.createElement('button');
  toggle.type = 'button'; toggle.id = 'catcul-cursor-toggle';
  toggle.innerHTML = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5 3 14 10-6 1-3 6L5 3Z"/><path d="m15 16 4 5"/></svg>';
  document.body.append(cursor, toggle);
  const supported = () => pointerMedia.matches && !contrastMedia.matches;
  const allowed = () => supported() && enabled && !dragging;
  function updateToggle() {
    toggle.classList.toggle('cc-toggle-ready', supported());
    toggle.setAttribute('aria-pressed', String(enabled));
    const label = enabled ? 'Use the standard cursor' : 'Use the Catcul custom cursor';
    toggle.title = label; toggle.setAttribute('aria-label', label);
  }
  function hide() {
    visible = false;
    root.classList.remove('catcul-cursor-active');
    cursor.classList.remove('cc-visible', 'cc-pressed');
  }
  function stateFor(element) {
    if (!(element instanceof Element)) return '';
    if (element.closest('[disabled], [aria-disabled="true"], fieldset:disabled')) return 'cc-disabled';
    const explicit = element.closest('[data-cursor]')?.getAttribute('data-cursor');
    if (explicit === 'text') return 'cc-text';
    if (explicit === 'zoom' || element.closest('[data-open-photo], .lightbox-image, img.photo')) return 'cc-media';
    if (explicit === 'pointer') return 'cc-control';
    const field = element.closest('input, textarea, [contenteditable="true"], [contenteditable=""], [role="textbox"]');
    if (field && !field.matches('input[type="button"],input[type="submit"],input[type="reset"],input[type="checkbox"],input[type="radio"],input[type="range"],input[type="color"],input[type="file"],input[type="image"],input[type="hidden"]')) return 'cc-text';
    if (element.closest('a[href], button, summary, select, label[for], input, audio, video, [role="button"], [role="link"], [role="tab"], [role="switch"], [data-action], [data-room], [data-filter], [data-react], [tabindex="0"]')) return 'cc-control';
    return '';
  }
  function state(element) {
    if (element instanceof Element && element.closest('iframe')) { hide(); return; }
    const kind = stateFor(element);
    ['cc-control', 'cc-text', 'cc-media', 'cc-disabled'].forEach(name => cursor.classList.toggle(name, name === kind));
  }
  function paint(event) {
    if (event.pointerType && event.pointerType !== 'mouse') { hide(); return; }
    if (!allowed()) { hide(); return; }
    x = event.clientX; y = event.clientY;
    if (x < 0 || y < 0 || x >= innerWidth || y >= innerHeight) { hide(); return; }
    // Track exactly, without a trailing animation that could obscure the click target.
    cursor.style.transform = 'translate3d(' + x + 'px,' + y + 'px,0)';
    visible = true;
    cursor.classList.add('cc-visible');
    root.classList.add('catcul-cursor-active');
    state(event.target);
  }
  document.addEventListener('pointermove', paint, {passive: true});
  document.addEventListener('pointerover', paint, {passive: true});
  document.addEventListener('pointerdown', event => {
    if (event.pointerType !== 'mouse') { hide(); return; }
    paint(event);
    if (visible) cursor.classList.add('cc-pressed');
  }, {passive: true});
  document.addEventListener('pointerup', () => cursor.classList.remove('cc-pressed'), {passive: true});
  document.addEventListener('pointercancel', hide, {passive: true});
  document.documentElement.addEventListener('pointerleave', hide, {passive: true});
  window.addEventListener('blur', hide);
  window.addEventListener('pagehide', hide);
  document.addEventListener('visibilitychange', () => { if (document.hidden) hide(); });
  document.addEventListener('keydown', event => { if (event.key === 'Tab') hide(); });
  document.addEventListener('dragstart', () => { dragging = true; hide(); });
  document.addEventListener('dragend', () => { dragging = false; });
  window.addEventListener('scroll', () => {
    if (!visible || frame) return;
    frame = requestAnimationFrame(() => { frame = 0; if (visible) state(document.elementFromPoint(x, y)); });
  }, {capture: true, passive: true});
  toggle.addEventListener('click', event => {
    enabled = !enabled;
    try { localStorage.setItem(storageKey, enabled ? 'on' : 'off'); } catch (_) {}
    updateToggle();
    if (!enabled) hide();
    else if (event.detail > 0) paint({clientX: event.clientX, clientY: event.clientY, pointerType: 'mouse', target: toggle});
  });
  function preferenceChanged() { updateToggle(); hide(); }
  pointerMedia.addEventListener('change', preferenceChanged);
  contrastMedia.addEventListener('change', preferenceChanged);
  window.addEventListener('storage', event => {
    if (event.key === storageKey || event.key === null) { enabled = event.newValue !== 'off'; updateToggle(); hide(); }
  });
  updateToggle();
})();
</script>
